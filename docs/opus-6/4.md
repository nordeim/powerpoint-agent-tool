#!/bin/bash
# Stage 0a: Initial Classification

REQUEST_TEXT="$1"
ESTIMATED_SLIDES="${2:-5}"  # Default estimate

# Count destructive keywords
DESTRUCTIVE_COUNT=0
for keyword in "delete" "remove" "erase" "clear"; do
  COUNT=$(echo "$REQUEST_TEXT" | grep -oi "$keyword" | wc -l)
  DESTRUCTIVE_COUNT=$((DESTRUCTIVE_COUNT + COUNT))
done

# Count mass operation keywords
MASS_COUNT=0
for keyword in "replace all" "mass" "global" "all slides" "entire"; do
  COUNT=$(echo "$REQUEST_TEXT" | grep -oi "$keyword" | wc -l)
  MASS_COUNT=$((MASS_COUNT + COUNT))
done

# Count merge keywords
MERGE_COUNT=0
for keyword in "merge" "combine" "consolidate"; do
  COUNT=$(echo "$REQUEST_TEXT" | grep -oi "$keyword" | wc -l)
  MERGE_COUNT=$((MERGE_COUNT + COUNT))
done

# Calculate initial score
SLIDE_SCORE=$(echo "$ESTIMATED_SLIDES * 0.3" | bc)
DESTRUCTIVE_SCORE=$(echo "$DESTRUCTIVE_COUNT * 3.0" | bc)
MASS_SCORE=$(echo "$MASS_COUNT * 3.0" | bc)
MERGE_SCORE=$(echo "$MERGE_COUNT * 1.5" | bc)

INITIAL_SCORE=$(echo "$SLIDE_SCORE + $DESTRUCTIVE_SCORE + $MASS_SCORE + $MERGE_SCORE" | bc)

# Classify
if (( $(echo "$INITIAL_SCORE < 3.0" | bc -l) )); then
  INITIAL_CLASS="SIMPLE"
elif (( $(echo "$INITIAL_SCORE <= 10.0" | bc -l) )); then
  INITIAL_CLASS="STANDARD"
else
  INITIAL_CLASS="COMPLEX"
fi

# Check for destructive flag
HAS_DESTRUCTIVE="false"
if [[ $DESTRUCTIVE_COUNT -gt 0 ]] || [[ $MASS_COUNT -gt 0 ]]; then
  HAS_DESTRUCTIVE="true"
fi

echo "Initial Classification: $INITIAL_CLASS (Score: $INITIAL_SCORE)"
echo "Destructive Flag: $HAS_DESTRUCTIVE"

4.3 6×6 Rule Validation
MANDATORY: Validate content against 6×6 rule before adding bullet lists.

Bash

#!/bin/bash
# 6x6 Rule Validation Script

validate_6x6_rule() {
  local ITEMS="$1"
  local ERRORS=0
  
  # Count items (comma-separated)
  ITEM_COUNT=$(echo "$ITEMS" | tr ',' '\n' | wc -l)
  
  if [[ $ITEM_COUNT -gt 6 ]]; then
    echo "❌ 6×6 VIOLATION: $ITEM_COUNT items exceeds maximum 6"
    echo "💡 Recommendation: Split across multiple slides"
    ERRORS=$((ERRORS + 1))
  fi
  
  # Check word count per item
  MAX_WORDS=0
  LONG_ITEMS=""
  
  while IFS=',' read -ra ITEM_ARRAY; do
    for item in "${ITEM_ARRAY[@]}"; do
      WORD_COUNT=$(echo "$item" | wc -w)
      if [[ $WORD_COUNT -gt 6 ]]; then
        LONG_ITEMS="$LONG_ITEMS\n  - \"$item\" ($WORD_COUNT words)"
      fi
      if [[ $WORD_COUNT -gt $MAX_WORDS ]]; then
        MAX_WORDS=$WORD_COUNT
      fi
    done
  done <<< "$ITEMS"
  
  if [[ $MAX_WORDS -gt 6 ]]; then
    echo "❌ 6×6 VIOLATION: Maximum $MAX_WORDS words per bullet exceeds limit of 6"
    echo "💡 Items exceeding limit:$LONG_ITEMS"
    echo "💡 Recommendation: Shorten bullets or move detail to speaker notes"
    ERRORS=$((ERRORS + 1))
  fi
  
  if [[ $ERRORS -eq 0 ]]; then
    echo "✅ 6×6 Rule: PASSED ($ITEM_COUNT items, max $MAX_WORDS words per bullet)"
    return 0
  else
    return 1
  fi
}

6.2 Typography System (v3.8 - Unified 12pt Minimum)
Font Size Scale (Points)
Element	Minimum	Recommended	Maximum	Notes
Main Title	36pt	44pt	60pt	Slide title on title slides
Slide Title	28pt	32pt	40pt	Standard slide headers
Subtitle	20pt	24pt	28pt	Supporting title text
Body Text	12pt	18pt	24pt	Main content
Bullet Points	12pt	16pt	20pt	List items
Captions	12pt	14pt	16pt	Image/chart captions
Footer/Legal	12pt	12pt	14pt	Bottom of slide
⚠️ CRITICAL: 12pt is the absolute minimum for ALL text. No exceptions permitted without explicit documentation.

12pt Minimum Rationale
Projection Clarity: Text below 12pt becomes illegible when projected
Accessibility: WCAG guidelines recommend minimum readable sizes
Universal Readability: Ensures content accessible to all audience members
Print Compatibility: Maintains readability in printed handouts
Exception Documentation Requirements
If font size below 12pt is absolutely required (extremely rare):

Document in manifest design_decisions with explicit business justification
Include accessibility impact assessment explaining why it's acceptable
Provide alternative access methods:
Speaker notes with full text
Handout with larger text
Alt text for screen readers
Obtain explicit user approval with notation in manifest
Flag for accessibility review during validation
